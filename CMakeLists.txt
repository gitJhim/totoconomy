cmake_minimum_required(VERSION 3.20)
project(totoconomy VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ECON_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
add_library(econ_core src/core/model.cpp src/core/policies.cpp src/core/simulation.cpp src/core/storage.cpp)
target_include_directories(econ_core PUBLIC include)
target_include_directories(econ_core SYSTEM PUBLIC third_party)
target_compile_options(econ_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
file(GLOB_RECURSE ECON_SOURCES CONFIGURE_DEPENDS src/*.cpp include/*.hpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${ECON_SOURCES})
set(ECON_SOURCE_HASHES "")
foreach(source IN LISTS ECON_SOURCES)
  file(SHA256 "${source}" source_hash)
  string(APPEND ECON_SOURCE_HASHES "${source_hash}")
endforeach()
string(SHA256 ECON_SOURCE_ID "${ECON_SOURCE_HASHES}")
string(SUBSTRING "${ECON_SOURCE_ID}" 0 16 ECON_SOURCE_ID)
target_compile_definitions(econ_core PUBLIC ECON_BUILD="${PROJECT_VERSION}-${ECON_SOURCE_ID}-${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}-${CMAKE_BUILD_TYPE}")
if(ECON_SANITIZE)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Sanitizers require GCC or Clang")
  endif()
  target_compile_options(econ_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(econ_core PUBLIC -fsanitize=address,undefined)
endif()
add_library(econ_tui src/tui/data.cpp src/tui/view.cpp src/tui/terminal.cpp)
target_link_libraries(econ_tui PUBLIC econ_core)
target_compile_options(econ_tui PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(econ src/cli/main.cpp)
target_link_libraries(econ PRIVATE econ_tui)
enable_testing()
add_executable(econ_tests tests/tests.cpp)
target_link_libraries(econ_tests PRIVATE econ_core)
target_compile_definitions(econ_tests PRIVATE ECON_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME core_acceptance COMMAND econ_tests)
add_test(NAME cli_help COMMAND econ --help)
add_test(NAME cli_version COMMAND econ --version)
add_test(NAME cli_validate COMMAND econ validate --scenario ${CMAKE_CURRENT_SOURCE_DIR}/scenarios/accounting_cycle.json --json)
add_test(NAME cli_acceptance COMMAND ${CMAKE_COMMAND} -DECON=$<TARGET_FILE:econ> -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR} -DOUT=${CMAKE_CURRENT_BINARY_DIR}/acceptance -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/acceptance.cmake)

add_executable(econ_tui_tests tests/tui_tests.cpp)
target_link_libraries(econ_tui_tests PRIVATE econ_tui util)
target_compile_definitions(econ_tui_tests PRIVATE ECON_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME tui_acceptance COMMAND econ_tui_tests $<TARGET_FILE:econ>)
